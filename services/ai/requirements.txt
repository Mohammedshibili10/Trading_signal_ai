# Core analysis service. Deliberately light so `docker compose up` finishes in
# under two minutes. Heavy ML extras live in requirements-ml.txt.
#
# Lower bounds rather than exact pins on the numeric stack. Exact pins look
# reproducible but break the moment you run a Python the pinned wheel wasn't
# built for — pip then falls back to compiling numpy/scipy from source, which
# either takes 20+ minutes or fails outright without a C toolchain. Ranges let
# pip pick the newest wheel that exists for whatever interpreter is present
# (3.11 through 3.14). The Dockerfile pins the interpreter, which is where
# reproducibility actually belongs.

fastapi>=0.115.6,<1.0
uvicorn[standard]>=0.34.0,<1.0
pydantic>=2.10.5,<3.0
pydantic-settings>=2.7.1,<3.0
python-multipart>=0.0.20

# Numerics. Every indicator, pattern and statistic in engine/ is implemented
# directly against these two — no TA library at runtime, so there is nothing to
# break when numpy majors bump.
numpy>=2.1
pandas>=2.2
scipy>=1.14

# Classical ML for the optional model-assisted scorers
scikit-learn>=1.5

httpx>=0.28.1,<1.0
# Verify TLS against the OS trust store rather than certifi's bundle. Without
# it, any machine behind a TLS-inspecting proxy or antivirus fails every
# outbound call with CERTIFICATE_VERIFY_FAILED while curl on the same box works.
truststore>=0.10.0
redis>=5.2.1,<7.0
orjson>=3.10.14
python-dateutil>=2.9.0
